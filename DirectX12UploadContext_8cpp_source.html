113void DirectX12UploadContext::Submit() {
114 if (!m_recording) return;
115 m_recording = false;
116
117 HRESULT hr = m_commandList->Close();
118 if (FAILED(hr)) {
119 SLEAK_ERROR("Failed to close upload command list! HRESULT: 0x{:08X}",
120 static_cast<unsigned int>(hr));
121 m_open = Batch{};
122 return;
123 }
124
125 ID3D12CommandList* lists[] = {m_commandList.Get()};
126 m_queue->ExecuteCommandLists(1, lists);
127 m_open.fenceValue = ++m_fenceValue;
128 m_queue->Signal(m_fence.Get(), m_fenceValue);
129
130 m_inFlight.push_back(std::move(m_open));
131 m_open = Batch{};
132}

144void DirectX12UploadContext::Retire() {
145 if (!m_fence) return;
146 const uint64_t completed = m_fence->GetCompletedValue();
147
148 size_t finished = 0;
149 while (finished < m_inFlight.size() &&
150 m_inFlight[finished].fenceValue <= completed) {
151 m_freeAllocators.push_back(std::move(m_inFlight[finished].allocator));
152 ++finished;
153 }
154 m_inFlight.erase(m_inFlight.begin(), m_inFlight.begin() + finished);
155}
