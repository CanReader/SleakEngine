164 SerializationFormat SerializationFactory::DetectFormat(const std::string& filePath) {
165 if (filePath.ends_with(".bin")) {
166 return SerializationFormat::Binary;
167 } else if (filePath.ends_with(".json")) {
168 return SerializationFormat::JSON;

170 return SerializationFormat::YAML;
