39 /// PBR-ish surface properties, textures, and GPU buffer for a drawable;
40 /// shared across any mesh that references it.
41 ///
42 /// A Material describes how a surface responds to light: a base color,
43 /// metallic, roughness, ambient occlusion, emissive, and opacity, plus
44 /// up to seven texture maps that override those scalars per texel. It
45 /// owns its own GPU constant buffer and rebuilds it when properties
46 /// change.
47 ///
48 /// Materials are meant to be shared. Wrap one in a RefPtr and hand the
49 /// same RefPtr to every MaterialComponent that should use it; changing
50 /// a property updates every object drawn with it. Textures are held by
51 /// RefPtr too. Each texture setter takes a `RefPtr<Texture>` to share
52 /// one texture between materials, a raw `Texture*` the material adopts
53 /// (do not hand the same raw pointer to two materials), or a file path
54 /// the material loads for you.
55 ///
56 /// SetRenderMode() decides how transparency is handled:
57 /// MaterialRenderMode::Opaque and ::Cutout render in the normal
58 /// (deferred, where the backend supports it) path, while ::Transparent
59 /// forces the material into a forward pass. IsForwardRendered()
60 /// reports which side a material falls on.
61 ///
62 /// @code{.cpp}
63 /// auto* mat = new Sleak::Material();
64 /// mat->SetShader("assets/shaders/default_shader.hlsl");
65 /// mat->SetDiffuseColor((uint8_t)230, (uint8_t)230, (uint8_t)230);
66 /// mat->SetMetallic(0.0f);
67 /// mat->SetRoughness(0.35f);
68 /// mat->SetAO(1.0f);
69 ///
70 /// // Texture maps, loaded by path
71 /// mat->SetDiffuseTexture("assets/textures/crate_albedo.png");
72 /// mat->SetNormalTexture("assets/textures/crate_normal.png");
73 /// mat->SetTiling(2.0f, 2.0f);
74 ///
75 /// // Share one material across many objects
76 /// Sleak::RefPtr<Sleak::Material> shared(mat);
77 /// crate->AddComponent<Sleak::MaterialComponent>(shared);
78 /// barrel->AddComponent<Sleak::MaterialComponent>(shared);
79 ///
80 /// // Alpha-blended glass
81 /// auto* glass = new Sleak::Material();
82 /// glass->SetRenderMode(Sleak::MaterialRenderMode::Transparent);
83 /// glass->SetOpacity(0.35f);
84 /// glass->SetTwoSided(true);
85 /// @endcode
86 ///
87 /// @see MaterialComponent, Texture, MaterialRenderMode, MeshBatch,
88 /// RefPtr
