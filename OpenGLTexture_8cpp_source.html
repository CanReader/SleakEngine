69void OpenGLTexture::SetFilter(TextureFilter filter) {
70 m_filter = filter;
71 glBindTexture(GL_TEXTURE_2D, m_texture);
72 switch (filter) {
73 case TextureFilter::Nearest:
74 glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_MIN_FILTER, GL_NEAREST_MIPMAP_NEAREST);
75 glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_MAG_FILTER, GL_NEAREST);
76 break;
77 case TextureFilter::Bilinear:
78 glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_MIN_FILTER, GL_LINEAR_MIPMAP_NEAREST);
79 glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_MAG_FILTER, GL_LINEAR);
80 break;
81 case TextureFilter::Trilinear:
82 glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_MIN_FILTER, GL_LINEAR_MIPMAP_LINEAR);
83 glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_MAG_FILTER, GL_LINEAR);
84 break;
85 case TextureFilter::Anisotropic2x:
86 case TextureFilter::Anisotropic4x:
87 case TextureFilter::Anisotropic8x:
88 case TextureFilter::Anisotropic16x: {
89 glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_MIN_FILTER, GL_LINEAR_MIPMAP_LINEAR);
90 glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_MAG_FILTER, GL_LINEAR);
91 float aniso = 1.0f;
92 if (filter == TextureFilter::Anisotropic2x) aniso = 2.0f;
93 else if (filter == TextureFilter::Anisotropic4x) aniso = 4.0f;
94 else if (filter == TextureFilter::Anisotropic8x) aniso = 8.0f;
95 else if (filter == TextureFilter::Anisotropic16x) aniso = 16.0f;
96 float maxSupported;
97 glGetFloatv(GL_MAX_TEXTURE_MAX_ANISOTROPY, &maxSupported);
98 glTexParameterf(GL_TEXTURE_2D, GL_TEXTURE_MAX_ANISOTROPY,
99 aniso < maxSupported ? aniso : maxSupported);
100 break;
101 }
102 default:
103 glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_MIN_FILTER, GL_LINEAR_MIPMAP_LINEAR);
104 glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_MAG_FILTER, GL_LINEAR);
105 break;
106 }
107 glBindTexture(GL_TEXTURE_2D, 0);
108}
