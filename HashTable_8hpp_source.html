12 * @class HashTable
13 * @brief Implements a hash table (or map) for efficient key-value storage and
14 * retrieval.
15 *
16 * The HashTable class provides fast access to values based on their associated
17 * keys. It is suitable for scenarios where you need to quickly look up data
18 * using unique identifiers. Use this class when you need to store and retrieve
19 * data based on keys, such as symbol tables, dictionaries, or caching systems.
20 *
21 * Example Use Cases:
22 * - Implementing a symbol table in a compiler.
23 * - Creating a dictionary or vocabulary lookup system.
24 * - Caching frequently accessed data.
25 * - Storing user preferences or configuration settings.
26 *
27 * Implementation Details:
28 * - Uses a hash function to map keys to indices in an internal array.
29 * - Handles collisions with linear-probing open addressing and tombstones.
30 * - Provides methods for inserting, deleting, and retrieving key-value pairs.
31 * @ingroup utility
32 */
33template <typename Key, typename Value>
34class HashTable {
35 private:
36 /// One slot: key/value plus occupied and tombstone-deleted flags.
37 struct Entry {

39 Value value;
40 bool occupied;

58 /// Home slot of key in a table with tableCapacity slots.
59 static size_t slotFor(const Key& key, size_t tableCapacity) {
60 return std::hash<Key>{}(key) % tableCapacity;
61 }
62
63 /// Home slot of key in the current table.
64 size_t hash(const Key& key) const { return slotFor(key, capacity); }
65
66 /// Slot holding a live entry for key, or capacity if there is none.
67 size_t findSlot(const Key& key) const {
68 size_t index = hash(key);
69 for (size_t probes = 0; probes < capacity && table[index].occupied;
70 ++probes) {
71 if (!table[index].deleted && table[index].key == key) {
72 return index;
73 }
74 index = (index + 1) % capacity;
75 }
76 return capacity;
77 }
78
79 /// Moves live entries into a newCapacity table, dropping tombstones.
80 void rehash(size_t newCapacity) {
81 Entry* newTable = new Entry[newCapacity];
82
83 for (size_t i = 0; i < capacity; ++i) {
84 if (table[i].occupied && !table[i].deleted) {
85 size_t index = slotFor(table[i].key, newCapacity);
86 while (newTable[index].occupied) {
87 index = (index + 1) % newCapacity;
88 }
89 newTable[index] = std::move(table[i]);
90 }
91 }
92
93 delete[] table;
94 table = newTable;
95 capacity = newCapacity;
96 tombstones = 0;
97 }
98
99 /// Grows if live entries fill the table, else purges tombstones only.
100 void resize() {
101 const bool mostlyLive = size * GROWTH_FACTOR >= capacity * loadFactor;
102 rehash(mostlyLive ? capacity * GROWTH_FACTOR : capacity);
103 }

144 /// Inserts or updates the value for key, resizing first if over the load
145 /// factor.
146 void insert(const Key& key, const Value& value) {
147 const size_t existing = findSlot(key);
148 if (existing != capacity) {
149 table[existing].value = value;
150 return;
151 }
152
153 if (size + tombstones >= capacity * loadFactor) {
154 resize();
155 }
156
157 size_t index = hash(key);
158 while (table[index].occupied && !table[index].deleted) {
159 index = (index + 1) % capacity;
160 }
161
162 if (table[index].deleted) {
163 --tombstones;
164 }
165 table[index].key = key;
166 table[index].value = value;
167 table[index].occupied = true;
168 table[index].deleted = false;
169 ++size;
170 }

176 return false;

178 table[index].key = Key();
