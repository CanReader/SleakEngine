106 Application::~Application() {
107 // Flush the GPU before tearing down any scene-owned resources (textures,
108 // meshes, materials) — the renderer's descriptor sets still reference
109 // samplers/image views owned by the game's scene objects. Without this
110 // wait, Texture destructors call vkDestroySampler while the descriptor
111 // set binding still has the sampler in use (VUID-vkDestroySampler-sampler-01082).
112 if (renderer) renderer->WaitIdle();
113
114 delete Game;
115 Sleak::UI::ShutdownTextureCache(); // frees cached VkImage/memory pre-device-teardown
116 Sleak::MeshBatch::Shutdown();
117 delete m_benchmark;
118 m_benchmark = nullptr;
119 delete m_DebugOverlay;
120 m_DebugOverlay = nullptr;
121
122 if (renderer)
123 renderer->Cleanup();
124
125 delete renderer;
126 delete CoreWindow;
127 SLEAK_LOG("The application has been successfully closed, have a good day sir");
128 }

130 int Application::Run(GameBase* game) {
131 Game = game;
132
133 float lastTime = FrameTimer.Elapsed();
134
135 if(CoreWindow && CoreWindow->InitializeWindow()) {
136
137 if(renderer && renderer->Initialize()) {
138
139 renderer->CreateImGUI();
140 if (renderer->GetImGUIEnabled())
141 CoreWindow->SetImGuiReady(true);
142
143 m_DebugOverlay = new DebugOverlay();
144 m_DebugOverlay->Initialize(renderer, game);
145
146 m_benchmark = new Benchmark();
147 m_benchmark->Initialize(renderer);
148 renderer->RegisterBenchmarkMetrics(*m_benchmark);
149 m_benchmark->RegisterMetric("CPU_frame_ms",
150 [this]() { return m_cpuFrameMs; });
151 {
152 auto& cfg = m_DebugOverlay->GetConfig();
153 cfg.ShowCameraPanel = false;
154 cfg.ShowPerformancePanel = false;
155 }
156
157 auto context = renderer->GetContext();
158 auto queue = RenderEngine::RenderCommandQueue::GetInstance();
159
160 float lastTime = FrameTimer.Elapsed();
161 float accumulator = 0.0f;
162 const float fixedTimestep = 1.0f / 60.0f;
163
164 // Frame capture
165 const std::string capturePath = CommandLine::GetValue("-capture");
166 const bool captureMode = !capturePath.empty();
167 uint64_t captureFrame = DEFAULT_CAPTURE_FRAME;
168 const std::string frameArg =
169 CommandLine::GetValue("-capture-frame");
170 if (!frameArg.empty()) {
171 try {
172 captureFrame = std::stoull(frameArg);
173 } catch (...) {
174 }
175 }
176 if (captureMode)
177 SLEAK_INFO(
178 "Frame capture: frame {} -> {} (fixed 1/60 s step, "
179 "resizes ignored)",
180 captureFrame, capturePath);
181 uint64_t frameIndex = 0;
182 bool captureArmed = false;
183
184 Debug::ShaderHotReload::AddSourceDirectory(SLEAK_ENGINE_SHADER_DIR);
185
186 // Initialize and begin the game (and scene)
187 if (Game) {
188 if (!Game->Initialize()) {
189 SLEAK_FATAL("Game failed to initialize!");
190 return -1;
191 }
192
193 Game->Begin();
194
195 // Apply CLI graphics settings
196 {
197 if (CommandLine::HasFlag("--vsync")) renderer->SetVSync(true);
198 if (CommandLine::HasFlag("--no-vsync")) renderer->SetVSync(false);
199
200 const std::string msaaStr = CommandLine::GetValue("-msaa");
201 if (!msaaStr.empty()) {
202 try { renderer->SetMSAASampleCount(
203 static_cast<uint32_t>(std::stoi(msaaStr))); }
204 catch (...) {}
205 }
206
207 if (CommandLine::HasFlag("--fullscreen")) CoreWindow->ToggleFullScreen();
208 }
209
210 // Auto-start benchmark if --bench / --benchmark was passed
211 if ((CommandLine::HasFlag("--bench") || CommandLine::HasFlag("--benchmark"))
212 && m_benchmark)
213 m_benchmark->ToggleRecording();
214 }
215
216 while(!CoreWindow->ShouldClose()) {
217
218 float currentTime = FrameTimer.Elapsed();
219 DeltaTime =
220 captureMode ? fixedTimestep : currentTime - lastTime;
221 lastTime = currentTime;
222
223 #if defined(_DEBUG) && defined(COUNT_FRAME)
224 ScopedTimer("Frame Timer");
225 #endif
226
227 CoreWindow->Update();
228
229 // Apply any pending resize (deferred from event handler to avoid GPU hang)
230 if (m_pendingResize && !captureMode) {
231 renderer->Resize(m_pendingResizeW, m_pendingResizeH);
232 width = static_cast<int>(m_pendingResizeW);
233 height = static_cast<int>(m_pendingResizeH);
234 m_pendingResize = false;
235
236 // Update active camera's projection matrix for the new aspect ratio.
237 // Vulkan renderer ignores the width/height args to Resize() and uses
238 // the Vulkan surface caps, so the camera must be notified separately.
239 if (Game && Game->GetActiveScene()) {
240 if (auto* cam = Game->GetActiveScene()->GetActiveCamera())
241 cam->OnResize(m_pendingResizeW, m_pendingResizeH);
242 }
243 }
244
245 if (captureMode && !captureArmed &&
246 frameIndex >= captureFrame) {
247 captureArmed = true;
248 if (!renderer->RequestFrameCapture(capturePath)) {
249 SLEAK_ERROR("Frame capture unavailable on {}",
250 renderer->GetTypeStr());
251 CoreWindow->Close();
252 }
253 }
254
255 Debug::ShaderHotReload::Poll();
256
257 renderer->BeginRender();
258 const auto cpuFrameStart = std::chrono::steady_clock::now();
259
260 // Update active scene if present
261 if (Game && Game->GetActiveScene()) {
262 auto* activeScene = Game->GetActiveScene();
263
264 // Fixed timestep updates (physics, etc.)
265 accumulator += DeltaTime;
266 while (accumulator >= fixedTimestep) {
267 activeScene->FixedUpdate(fixedTimestep);
268 accumulator -= fixedTimestep;
269 }
270
271 // Per-frame update
272 activeScene->Update(DeltaTime);
273
274 // Late update (after all updates, e.g. camera follow)
275 activeScene->LateUpdate(DeltaTime);
276 }
277
278 // Per-frame game logic
279 if (Game)
280 Game->Loop(DeltaTime);
281 if (m_DebugOverlay)
282 m_DebugOverlay->Render(DeltaTime);
283
284 if (m_benchmark)
285 m_benchmark->Tick(DeltaTime);
286
287 renderer->FlushPendingTransfers();
288
289 if (queue && context)
290 queue->ExecuteCommands(context);
291
292 m_cpuFrameMs =
293 std::chrono::duration<float, std::milli>(
294 std::chrono::steady_clock::now() - cpuFrameStart)
295 .count();
296 renderer->EndRender();
297
298 if (captureArmed && !renderer->IsFrameCapturePending())
299 CoreWindow->Close();
300 ++frameIndex;
301 }
302 }
303 else{
304 SLEAK_FATAL("Unable to initialize graphics!");
305 return -1;
306 }
307
308 }
309 else {
310 SLEAK_FATAL("App cannot run without any window!");
311 return -2;
312 }
313
314 return 0;
315 }

367 if (fpc) {

371 if (ctrl)
372 ctrl->SetEnabled(!ctrl->IsEnabled());

375 }

377 break;

380 CoreWindow->ToggleFullScreen();
381 break;
382
383 case Input::KEY_CODE::KEY__F12:
