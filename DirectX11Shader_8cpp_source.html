86bool DirectX11Shader::BuildStages(StageSources sources) {
87 Microsoft::WRL::ComPtr<ID3D11VertexShader> vertexShader;
88 Microsoft::WRL::ComPtr<ID3D11PixelShader> pixelShader;
89 Microsoft::WRL::ComPtr<ID3DBlob> vertexBlob;
90 Microsoft::WRL::ComPtr<ID3DBlob> pixelBlob;
91 if (!compileShader(sources.vertPath, sources.vertEntry, VERTEX_PROFILE,
92 vertexShader, vertexBlob)) {
93 SLEAK_ERROR("Failed to compile vertex shader {}", sources.vertPath);
94 return false;
95 }
96 if (!compileShader(sources.fragPath, sources.fragEntry, PIXEL_PROFILE,
97 pixelShader, pixelBlob)) {
98 SLEAK_ERROR("Failed to compile pixel shader {}", sources.fragPath);
99 return false;
100 }
101
102 // A rebuild needs a layout validated against the new vertex inputs
103 Microsoft::WRL::ComPtr<ID3D11InputLayout> layout;
104 if (Layout && !CreateLayoutFor(vertexBlob.Get(), layout)) return false;
105
106 m_vertexShader = std::move(vertexShader);
107 m_pixelShader = std::move(pixelShader);
108 m_vertexShaderBlob = std::move(vertexBlob);
109 m_pixelShaderBlob = std::move(pixelBlob);
110 if (layout) Layout = std::move(layout);
111#ifdef SLEAK_HOT_RELOAD
112 m_vertStem = StemFromPath(sources.vertPath);
113 m_fragStem = StemFromPath(sources.fragPath);
114 m_sources = std::move(sources);
115#endif
116 return true;
117}

145bool DirectX11Shader::CreateLayoutFor(
146 ID3DBlob* vertexBlob, Microsoft::WRL::ComPtr<ID3D11InputLayout>& layout) {
147 // Check via reflection if the shader uses BLENDINDICES (skinned shader)
148 bool isSkinned = false;
149 Microsoft::WRL::ComPtr<ID3D11ShaderReflection> reflector;
150 HRESULT hr =
151 D3DReflect(vertexBlob->GetBufferPointer(), vertexBlob->GetBufferSize(),
152 IID_ID3D11ShaderReflection,
153 reinterpret_cast<void**>(reflector.GetAddressOf()));
154 if (SUCCEEDED(hr)) {
155 D3D11_SHADER_DESC shaderDesc;
156 reflector->GetDesc(&shaderDesc);
157 for (UINT p = 0; p < shaderDesc.InputParameters; ++p) {
158 D3D11_SIGNATURE_PARAMETER_DESC paramDesc;
159 reflector->GetInputParameterDesc(p, &paramDesc);
160 if (strcmp(paramDesc.SemanticName, "BLENDINDICES") == 0) {
161 isSkinned = true;
162 break;
163 }
164 }
165 }
166
167 D3D11_INPUT_ELEMENT_DESC* layoutDesc = isSkinned ? SkinnedLayout : DefaultLayout;
168 UINT layoutCount = isSkinned ? ARRAYSIZE(SkinnedLayout) : ARRAYSIZE(DefaultLayout);
169
170 hr = m_device->CreateInputLayout(
171 layoutDesc, layoutCount, vertexBlob->GetBufferPointer(),
172 vertexBlob->GetBufferSize(), layout.ReleaseAndGetAddressOf());
173
174 if (FAILED(hr)) {
175 SLEAK_ERROR("Failed to create input layout: 0x{:08X}",
176 static_cast<unsigned int>(hr));
177 return false;
178 }
179 return true;
180}

182/// Compiles a shader stage from an HLSL file and creates the matching D3D11 shader object.
183template <typename T>
184bool DirectX11Shader::compileShader(const std::string& filePath,
185 const std::string& entryPoint,
186 const std::string& profile,
187 Microsoft::WRL::ComPtr<T>& shader,
188 Microsoft::WRL::ComPtr<ID3DBlob>& blob) {
189 Microsoft::WRL::ComPtr<ID3DBlob> errorBlob;
190 HRESULT hr = D3DCompileFromFile(
191 std::wstring(filePath.begin(), filePath.end()).c_str(), // File path
192 nullptr, // Defines
193 D3D_COMPILE_STANDARD_FILE_INCLUDE, // Includes
194 entryPoint.c_str(), // Entry point
195 profile.c_str(), // Shader profile (e.g., "vs_5_0")
196 D3DCOMPILE_DEBUG, // Flags
197 0, // Flags 2
198 &blob, // Output blob
199 &errorBlob // Error blob
200 );
201
202 if (FAILED(hr)) {
203 if (errorBlob) {
204 SLEAK_ERROR("Shader compilation error in {} ({}): {}", filePath,
205 entryPoint, (char*)errorBlob->GetBufferPointer());
206 } else {
207 SLEAK_ERROR("Shader compilation error in {} ({}): 0x{:08X}",
208 filePath, entryPoint, static_cast<unsigned int>(hr));
209 }
210 return false;
211 }
212
213 // Create the shader object
214 if constexpr (std::is_same_v<T, ID3D11VertexShader>) {
215 hr = m_device->CreateVertexShader(
216 blob->GetBufferPointer(), blob->GetBufferSize(), nullptr,
217 reinterpret_cast<ID3D11VertexShader**>(shader.GetAddressOf()));
218 } else if constexpr (std::is_same_v<T, ID3D11PixelShader>) {
219 hr = m_device->CreatePixelShader(
220 blob->GetBufferPointer(), blob->GetBufferSize(), nullptr,
221 reinterpret_cast<ID3D11PixelShader**>(shader.GetAddressOf()));
222 }
223
224 if (FAILED(hr)) {
225 SLEAK_ERROR("Failed to create shader object for {} ({}): 0x{:08X}",
226 filePath, entryPoint, static_cast<unsigned int>(hr));
227 }
228 return SUCCEEDED(hr);
229}
