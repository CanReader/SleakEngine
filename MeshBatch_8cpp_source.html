67/// Binds the batch material and (re)uploads the shared identity-world transform buffer once per batch.
68void MeshBatch::BeginBatch(Material* material) {
69 auto* queue = RenderEngine::RenderCommandQueue::GetInstance();
70
71 // Bind material once for all draws in this batch
72 queue->SubmitBindMaterial(material);
73
74 // Build identity world matrix — batch vertices are already in world space
75 auto world = Math::Matrix4::Identity();
76 RenderEngine::TransformBuffer tb(world,
77 Camera::GetMainViewMatrix(),
78 Camera::GetMainProjectionMatrix());
79
80 if (!s_batchTransformBuffer) {
81 s_batchTransformBuffer = RefPtr<RenderEngine::BufferBase>(
82 RenderEngine::ResourceManager::CreateBuffer(
83 RenderEngine::BufferType::Constant,
84 tb.GetSize(), tb.GetData()));
85 } else {
86 queue->SubmitUpdateConstantBuffer(
87 s_batchTransformBuffer, tb.GetData(), tb.GetSize());
88 }
89
90 queue->SubmitBindConstantBuffer(s_batchTransformBuffer, 0);
91}
