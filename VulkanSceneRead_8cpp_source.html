27/// True when the depth format carries a stencil aspect.
28bool HasStencil(VkFormat format) {
29 return format == VK_FORMAT_D32_SFLOAT_S8_UINT ||
30 format == VK_FORMAT_D24_UNORM_S8_UINT ||
31 format == VK_FORMAT_D16_UNORM_S8_UINT;
32}

34/// Aspect mask covering every aspect of a depth format.
35VkImageAspectFlags DepthAspects(VkFormat format) {
36 return VK_IMAGE_ASPECT_DEPTH_BIT |
37 (HasStencil(format) ? VK_IMAGE_ASPECT_STENCIL_BIT : 0);
38}

40/// Picks a memory type matching the filter and property flags.
41bool FindMemory(VkPhysicalDevice gpu, uint32_t typeBits,
42 VkMemoryPropertyFlags flags, uint32_t& index) {
43 VkPhysicalDeviceMemoryProperties props;
44 vkGetPhysicalDeviceMemoryProperties(gpu, &props);
45 for (uint32_t i = 0; i < props.memoryTypeCount; ++i) {
46 if ((typeBits & (1u << i)) &&
47 (props.memoryTypes[i].propertyFlags & flags) == flags) {
48 index = i;
49 return true;
50 }
51 }
52 return false;
53}

55/// Creates a device-local 2D image, or a cube when given six layers, and
56/// its view.
57bool CreateSampledImage(VkDevice device, VkPhysicalDevice gpu, VkFormat format,
58 VkExtent2D extent, uint32_t layers,
59 VkImageUsageFlags usage, VkImageAspectFlags aspect,
60 VkImage& image, VkDeviceMemory& memory,
61 VkImageView& view) {
62 const bool cube = layers == CUBE_FACE_COUNT;
63
64 VkImageCreateInfo info{};
65 info.sType = VK_STRUCTURE_TYPE_IMAGE_CREATE_INFO;
66 info.flags = cube ? VK_IMAGE_CREATE_CUBE_COMPATIBLE_BIT : 0;
67 info.imageType = VK_IMAGE_TYPE_2D;
68 info.extent = {extent.width, extent.height, 1};
69 info.mipLevels = 1;
70 info.arrayLayers = layers;
71 info.format = format;
72 info.tiling = VK_IMAGE_TILING_OPTIMAL;
73 info.initialLayout = VK_IMAGE_LAYOUT_UNDEFINED;
74 info.usage = usage;
75 info.samples = VK_SAMPLE_COUNT_1_BIT;
76 info.sharingMode = VK_SHARING_MODE_EXCLUSIVE;
77 if (vkCreateImage(device, &info, nullptr, &image) != VK_SUCCESS)
78 return false;
79
80 VkMemoryRequirements req;
81 vkGetImageMemoryRequirements(device, image, &req);
82 VkMemoryAllocateInfo alloc{};
83 alloc.sType = VK_STRUCTURE_TYPE_MEMORY_ALLOCATE_INFO;
84 alloc.allocationSize = req.size;
85 if (!FindMemory(gpu, req.memoryTypeBits,
86 VK_MEMORY_PROPERTY_DEVICE_LOCAL_BIT,
87 alloc.memoryTypeIndex) ||
88 vkAllocateMemory(device, &alloc, nullptr, &memory) != VK_SUCCESS)
89 return false;
90 vkBindImageMemory(device, image, memory, 0);
91
92 VkImageViewCreateInfo vinfo{};
93 vinfo.sType = VK_STRUCTURE_TYPE_IMAGE_VIEW_CREATE_INFO;
94 vinfo.image = image;
95 vinfo.viewType = cube ? VK_IMAGE_VIEW_TYPE_CUBE : VK_IMAGE_VIEW_TYPE_2D;
96 vinfo.format = format;
97 vinfo.subresourceRange = {aspect, 0, 1, 0, layers};
98 return vkCreateImageView(device, &vinfo, nullptr, &view) == VK_SUCCESS;
99}

112/// Image barrier over every layer of a single-mip image.

137/// Clears images to zero (depth to 1) and leaves them shader readable.
138void ClearToShaderRead(VkDevice device, VkCommandPool pool, VkQueue queue,
139 const ClearTarget* targets, uint32_t count) {
140 VkCommandBufferAllocateInfo ca{};
141 ca.sType = VK_STRUCTURE_TYPE_COMMAND_BUFFER_ALLOCATE_INFO;
142 ca.level = VK_COMMAND_BUFFER_LEVEL_PRIMARY;
143 ca.commandPool = pool;
144 ca.commandBufferCount = 1;
145 VkCommandBuffer cmd = VK_NULL_HANDLE;
146 if (vkAllocateCommandBuffers(device, &ca, &cmd) != VK_SUCCESS) return;
147
148 VkCommandBufferBeginInfo bi{};
149 bi.sType = VK_STRUCTURE_TYPE_COMMAND_BUFFER_BEGIN_INFO;
150 bi.flags = VK_COMMAND_BUFFER_USAGE_ONE_TIME_SUBMIT_BIT;
151 vkBeginCommandBuffer(cmd, &bi);
152
153 for (uint32_t i = 0; i < count; ++i) {
154 const ClearTarget& t = targets[i];
155 const VkImageMemoryBarrier toDst =
156 MakeBarrier(t.image, t.aspect, t.layers, VK_IMAGE_LAYOUT_UNDEFINED,
157 VK_IMAGE_LAYOUT_TRANSFER_DST_OPTIMAL, 0,
158 VK_ACCESS_TRANSFER_WRITE_BIT);
159 vkCmdPipelineBarrier(cmd, VK_PIPELINE_STAGE_TOP_OF_PIPE_BIT,
160 VK_PIPELINE_STAGE_TRANSFER_BIT, 0, 0, nullptr, 0,
161 nullptr, 1, &toDst);
162
163 const VkImageSubresourceRange range = {t.aspect, 0, 1, 0, t.layers};
164 if (t.aspect & VK_IMAGE_ASPECT_DEPTH_BIT) {
165 // windows.h defines `far` as an empty macro
166 const VkClearDepthStencilValue farDepth = {FAR_DEPTH, 0};
167 vkCmdClearDepthStencilImage(cmd, t.image,
168 VK_IMAGE_LAYOUT_TRANSFER_DST_OPTIMAL,
169 &farDepth, 1, &range);
170 } else {
171 const VkClearColorValue zero = {};
172 vkCmdClearColorImage(cmd, t.image,
173 VK_IMAGE_LAYOUT_TRANSFER_DST_OPTIMAL, &zero, 1,
174 &range);
175 }
176
177 const VkImageMemoryBarrier toRead = MakeBarrier(
178 t.image, t.aspect, t.layers, VK_IMAGE_LAYOUT_TRANSFER_DST_OPTIMAL,
179 VK_IMAGE_LAYOUT_SHADER_READ_ONLY_OPTIMAL,
180 VK_ACCESS_TRANSFER_WRITE_BIT, VK_ACCESS_SHADER_READ_BIT);
181 vkCmdPipelineBarrier(cmd, VK_PIPELINE_STAGE_TRANSFER_BIT,
182 VK_PIPELINE_STAGE_VERTEX_SHADER_BIT |
183 VK_PIPELINE_STAGE_FRAGMENT_SHADER_BIT,
184 0, 0, nullptr, 0, nullptr, 1, &toRead);
185 }
186 vkEndCommandBuffer(cmd);
187
188 VkSubmitInfo submit{};
189 submit.sType = VK_STRUCTURE_TYPE_SUBMIT_INFO;
190 submit.commandBufferCount = 1;
191 submit.pCommandBuffers = &cmd;
192 vkQueueSubmit(queue, 1, &submit, VK_NULL_HANDLE);
193 vkQueueWaitIdle(queue);
194 vkFreeCommandBuffers(device, pool, 1, &cmd);
195}

199/// Creates the scene-read samplers, stand-in images, and per-frame UBOs and
200/// writes them into both frames' set 3 so the bindings are always valid.
201bool VulkanRenderer::CreateSceneReadResources() {
202 if (m_sceneReadCreated) return true;
203 if (!m_lightUBOCreated) return false;
204
205 // Samplers
206 VkSamplerCreateInfo sinfo{};
207 sinfo.sType = VK_STRUCTURE_TYPE_SAMPLER_CREATE_INFO;
208 sinfo.magFilter = VK_FILTER_LINEAR;
209 sinfo.minFilter = VK_FILTER_LINEAR;
210 sinfo.mipmapMode = VK_SAMPLER_MIPMAP_MODE_NEAREST;
211 sinfo.addressModeU = VK_SAMPLER_ADDRESS_MODE_CLAMP_TO_EDGE;
212 sinfo.addressModeV = VK_SAMPLER_ADDRESS_MODE_CLAMP_TO_EDGE;
213 sinfo.addressModeW = VK_SAMPLER_ADDRESS_MODE_CLAMP_TO_EDGE;
214 sinfo.maxLod = 0.0f;
215 if (vkCreateSampler(device, &sinfo, nullptr, &m_sceneLinearSampler) !=
216 VK_SUCCESS)
217 return false;
218 sinfo.magFilter = VK_FILTER_NEAREST;
219 sinfo.minFilter = VK_FILTER_NEAREST;
220 if (vkCreateSampler(device, &sinfo, nullptr, &m_scenePointSampler) !=
221 VK_SUCCESS)
222 return false;
223
224 // Stand-ins
225 const VkImageUsageFlags standInUsage =
226 VK_IMAGE_USAGE_SAMPLED_BIT | VK_IMAGE_USAGE_TRANSFER_DST_BIT;
227 if (!CreateSampledImage(device, physicalDevice, VK_FORMAT_R8G8B8A8_UNORM,
228 STAND_IN_EXTENT, SINGLE_LAYER, standInUsage,
229 VK_IMAGE_ASPECT_COLOR_BIT, m_sceneFallbackImage,
230 m_sceneFallbackMemory, m_sceneFallbackView) ||
231 !CreateSampledImage(device, physicalDevice, VK_FORMAT_R8G8B8A8_UNORM,
232 STAND_IN_EXTENT, CUBE_FACE_COUNT, standInUsage,
233 VK_IMAGE_ASPECT_COLOR_BIT, m_sceneFallbackCubeImage,
234 m_sceneFallbackCubeMemory,
235 m_sceneFallbackCubeView)) {
236 SLEAK_ERROR("VulkanRenderer: scene-read stand-in images failed");
237 return false;
238 }
239 const ClearTarget standIns[] = {
240 {m_sceneFallbackImage, VK_IMAGE_ASPECT_COLOR_BIT, SINGLE_LAYER},
241 {m_sceneFallbackCubeImage, VK_IMAGE_ASPECT_COLOR_BIT, CUBE_FACE_COUNT}};
242 ClearToShaderRead(device, commands, graphicsQueue, standIns,
243 static_cast<uint32_t>(std::size(standIns)));
244
245 // Per-frame UBOs
246 for (uint32_t i = 0; i < MAX_FRAMES_IN_FLIGHT; ++i) {
247 VkBufferCreateInfo binfo{};
248 binfo.sType = VK_STRUCTURE_TYPE_BUFFER_CREATE_INFO;
249 binfo.size = sizeof(SceneReadUBO);
250 binfo.usage = VK_BUFFER_USAGE_UNIFORM_BUFFER_BIT;
251 binfo.sharingMode = VK_SHARING_MODE_EXCLUSIVE;
252 if (vkCreateBuffer(device, &binfo, nullptr,
253 &m_sceneReadUBOBuffers[i]) != VK_SUCCESS)
254 return false;
255
256 VkMemoryRequirements req;
257 vkGetBufferMemoryRequirements(device, m_sceneReadUBOBuffers[i], &req);
258 VkMemoryAllocateInfo alloc{};
259 alloc.sType = VK_STRUCTURE_TYPE_MEMORY_ALLOCATE_INFO;
260 alloc.allocationSize = req.size;
261 if (!FindMemory(physicalDevice, req.memoryTypeBits,
262 VK_MEMORY_PROPERTY_HOST_VISIBLE_BIT |
263 VK_MEMORY_PROPERTY_HOST_COHERENT_BIT,
264 alloc.memoryTypeIndex) ||
265 vkAllocateMemory(device, &alloc, nullptr,
266 &m_sceneReadUBOMemory[i]) != VK_SUCCESS)
267 return false;
268 vkBindBufferMemory(device, m_sceneReadUBOBuffers[i],
269 m_sceneReadUBOMemory[i], 0);
270 vkMapMemory(device, m_sceneReadUBOMemory[i], 0, sizeof(SceneReadUBO), 0,
271 &m_sceneReadUBOMapped[i]);
272 std::memset(m_sceneReadUBOMapped[i], 0, sizeof(SceneReadUBO));
273
274 VkDescriptorBufferInfo bufInfo{};
275 bufInfo.buffer = m_sceneReadUBOBuffers[i];
276 bufInfo.range = sizeof(SceneReadUBO);
277
278 VkWriteDescriptorSet write{};
279 write.sType = VK_STRUCTURE_TYPE_WRITE_DESCRIPTOR_SET;
280 write.dstSet = m_shadowSamplerDescriptorSets[i];
281 write.dstBinding = SCENE_UBO_BINDING;
282 write.descriptorCount = 1;
283 write.descriptorType = VK_DESCRIPTOR_TYPE_UNIFORM_BUFFER;
284 write.pBufferInfo = &bufInfo;
285 vkUpdateDescriptorSets(device, 1, &write, 0, nullptr);
286 }
287
288 m_sceneReadCreated = true;
289 m_sceneReadWritten = {};
290 for (uint32_t i = 0; i < MAX_FRAMES_IN_FLIGHT; ++i)
291 UpdateSceneReadDescriptors(i);
292 return true;
293}

295/// Destroys everything CreateSceneReadResources made plus the snapshots.
296void VulkanRenderer::CleanupSceneReadResources() {
297 CleanupSceneCopyImages();
298 for (uint32_t i = 0; i < MAX_FRAMES_IN_FLIGHT; ++i) {
299 if (m_sceneReadUBOMapped[i])
300 vkUnmapMemory(device, m_sceneReadUBOMemory[i]);
301 if (m_sceneReadUBOBuffers[i])
302 vkDestroyBuffer(device, m_sceneReadUBOBuffers[i], nullptr);
303 if (m_sceneReadUBOMemory[i])
304 vkFreeMemory(device, m_sceneReadUBOMemory[i], nullptr);
305 m_sceneReadUBOMapped[i] = nullptr;
306 m_sceneReadUBOBuffers[i] = VK_NULL_HANDLE;
307 m_sceneReadUBOMemory[i] = VK_NULL_HANDLE;
308 }
309 DestroyImage(device, m_sceneFallbackImage, m_sceneFallbackMemory,
310 m_sceneFallbackView);
311 DestroyImage(device, m_sceneFallbackCubeImage, m_sceneFallbackCubeMemory,
312 m_sceneFallbackCubeView);
313 if (m_sceneLinearSampler)
314 vkDestroySampler(device, m_sceneLinearSampler, nullptr);
315 if (m_scenePointSampler)
316 vkDestroySampler(device, m_scenePointSampler, nullptr);
317 m_sceneLinearSampler = VK_NULL_HANDLE;
318 m_scenePointSampler = VK_NULL_HANDLE;
319 m_sceneReadWritten = {};
320 m_sceneReadCreated = false;
321}

357/// Points a frame's set 3 bindings 2-4 at the current snapshot and sky.
358/// Call only while that frame's command buffer is not pending.
359void VulkanRenderer::UpdateSceneReadDescriptors(uint32_t frame) {
360 if (!m_sceneReadCreated) return;
361
362 SceneReadViews want;
363 want.color =
364 m_sceneColorCopyView ? m_sceneColorCopyView : m_sceneFallbackView;
365 want.depth =
366 m_sceneDepthCopyView ? m_sceneDepthCopyView : m_sceneFallbackView;
367 want.sky =
368 m_skyboxCubemapView ? m_skyboxCubemapView : m_sceneFallbackCubeView;
369 want.skySerial = m_skyboxViewSerial;
370
371 SceneReadViews& have = m_sceneReadWritten[frame];
372 std::array<VkDescriptorImageInfo, SCENE_READ_IMAGE_COUNT> infos{};
373 std::array<VkWriteDescriptorSet, SCENE_READ_IMAGE_COUNT> writes{};
374 uint32_t count = 0;
375 auto add = [&](uint32_t binding, VkImageView view, VkSampler sampler) {
376 infos[count].imageLayout = VK_IMAGE_LAYOUT_SHADER_READ_ONLY_OPTIMAL;
377 infos[count].imageView = view;
378 infos[count].sampler = sampler;
379 writes[count].sType = VK_STRUCTURE_TYPE_WRITE_DESCRIPTOR_SET;
380 writes[count].dstSet = m_shadowSamplerDescriptorSets[frame];
381 writes[count].dstBinding = binding;
382 writes[count].descriptorCount = 1;
383 writes[count].descriptorType =
384 VK_DESCRIPTOR_TYPE_COMBINED_IMAGE_SAMPLER;
385 writes[count].pImageInfo = &infos[count];
386 ++count;
387 };
388 if (have.color != want.color)
389 add(SCENE_COLOR_BINDING, want.color, m_sceneLinearSampler);
390 if (have.depth != want.depth)
391 add(SCENE_DEPTH_BINDING, want.depth, m_scenePointSampler);
392 if (have.sky != want.sky || have.skySerial != want.skySerial)
393 add(SCENE_SKY_BINDING, want.sky,
394 m_skyboxCubemapView ? m_skyboxCubemapSampler
395 : m_sceneLinearSampler);
396 if (count > 0)
397 vkUpdateDescriptorSets(device, count, writes.data(), 0, nullptr);
398 have = want;
399}

401/// Fills this frame's SceneReadUBO and, inside the deferred forward pass,
402/// snapshots the scene. Runs once per frame on the first scene-reading draw.
403void VulkanRenderer::PrepareSceneRead() {
404 if (m_sceneReadPrepared || !m_sceneReadCreated) return;
405 m_sceneReadPrepared = true;
406
407 const SceneReadViews& bound = m_sceneReadWritten[currentFrame];
408 const bool sceneValid = m_forwardPassOpen && m_inForwardTransparentPass &&
409 m_sceneColorCopyImage != VK_NULL_HANDLE &&
410 bound.color == m_sceneColorCopyView &&
411 bound.depth == m_sceneDepthCopyView;
412 if (sceneValid) RecordSceneCopy();
413
414 SceneReadUBO ubo{};
415 Math::Matrix4 view, projection;
416 std::memcpy(&view(0, 0), m_cachedView, sizeof(m_cachedView));
417 std::memcpy(&projection(0, 0), m_cachedProjection,
418 sizeof(m_cachedProjection));
419 const Math::Matrix4 viewProj = view * projection;
420 std::memcpy(ubo.ViewProj, &viewProj(0, 0), sizeof(ubo.ViewProj));
421 std::memcpy(ubo.InvViewProj, m_cachedInvViewProj, sizeof(ubo.InvViewProj));
422 ubo.ScreenWidth = static_cast<float>(std::max(scExtent.width, 1u));
423 ubo.ScreenHeight = static_cast<float>(std::max(scExtent.height, 1u));
424 ubo.InvScreenWidth = 1.0f / ubo.ScreenWidth;
425 ubo.InvScreenHeight = 1.0f / ubo.ScreenHeight;
426 // Same UV-to-NDC scale the GBuffer push constants use
427 const bool jitter =
428 m_taaEnabled && m_taaResourcesCreated && m_inForwardTransparentPass;
429 ubo.JitterX = jitter ? m_taaJitter[0] * UV_TO_NDC : 0.0f;
430 ubo.JitterY = jitter ? m_taaJitter[1] * UV_TO_NDC : 0.0f;
431 ubo.NearPlane = m_cachedNearPlane;
432 ubo.FarPlane = m_cachedFarPlane;
433 ubo.SceneValid = static_cast<float>(sceneValid);
434 ubo.SkyValid = static_cast<float>(bound.sky != m_sceneFallbackCubeView);
435 std::memcpy(m_sceneReadUBOMapped[currentFrame], &ubo, sizeof(ubo));
436}

438/// Ends the forward pass, copies the HDR scene and depth into the
439/// snapshots, and reopens the pass with its contents loaded.
440void VulkanRenderer::RecordSceneCopy() {
441 vkCmdEndRenderPass(command);
442
443 const VkImageAspectFlags depthAspects = DepthAspects(depthFormat);
444 const VkImageMemoryBarrier before[] = {
445 MakeBarrier(m_hdrSceneImage, VK_IMAGE_ASPECT_COLOR_BIT, SINGLE_LAYER,
446 VK_IMAGE_LAYOUT_SHADER_READ_ONLY_OPTIMAL,
447 VK_IMAGE_LAYOUT_TRANSFER_SRC_OPTIMAL,
448 VK_ACCESS_COLOR_ATTACHMENT_WRITE_BIT,
449 VK_ACCESS_TRANSFER_READ_BIT),
450 MakeBarrier(depthImage, depthAspects, SINGLE_LAYER,
451 VK_IMAGE_LAYOUT_DEPTH_STENCIL_ATTACHMENT_OPTIMAL,
452 VK_IMAGE_LAYOUT_TRANSFER_SRC_OPTIMAL,
453 VK_ACCESS_DEPTH_STENCIL_ATTACHMENT_WRITE_BIT,
454 VK_ACCESS_TRANSFER_READ_BIT),
455 MakeBarrier(m_sceneColorCopyImage, VK_IMAGE_ASPECT_COLOR_BIT,
456 SINGLE_LAYER, VK_IMAGE_LAYOUT_UNDEFINED,
457 VK_IMAGE_LAYOUT_TRANSFER_DST_OPTIMAL, 0,
458 VK_ACCESS_TRANSFER_WRITE_BIT),
459 MakeBarrier(m_sceneDepthCopyImage, depthAspects, SINGLE_LAYER,
460 VK_IMAGE_LAYOUT_UNDEFINED,
461 VK_IMAGE_LAYOUT_TRANSFER_DST_OPTIMAL, 0,
462 VK_ACCESS_TRANSFER_WRITE_BIT)};
463 vkCmdPipelineBarrier(command,
464 VK_PIPELINE_STAGE_COLOR_ATTACHMENT_OUTPUT_BIT |
465 VK_PIPELINE_STAGE_EARLY_FRAGMENT_TESTS_BIT |
466 VK_PIPELINE_STAGE_LATE_FRAGMENT_TESTS_BIT |
467 VK_PIPELINE_STAGE_FRAGMENT_SHADER_BIT,
468 VK_PIPELINE_STAGE_TRANSFER_BIT, 0, 0, nullptr, 0,
469 nullptr, static_cast<uint32_t>(std::size(before)),
470 before);
471
472 VkImageCopy region{};
473 region.srcSubresource = {VK_IMAGE_ASPECT_COLOR_BIT, 0, 0, 1};
474 region.dstSubresource = region.srcSubresource;
475 region.extent = {scExtent.width, scExtent.height, 1};
476 vkCmdCopyImage(command, m_hdrSceneImage,
477 VK_IMAGE_LAYOUT_TRANSFER_SRC_OPTIMAL, m_sceneColorCopyImage,
478 VK_IMAGE_LAYOUT_TRANSFER_DST_OPTIMAL, 1, &region);
479 region.srcSubresource.aspectMask = VK_IMAGE_ASPECT_DEPTH_BIT;
480 region.dstSubresource.aspectMask = VK_IMAGE_ASPECT_DEPTH_BIT;
481 vkCmdCopyImage(command, depthImage, VK_IMAGE_LAYOUT_TRANSFER_SRC_OPTIMAL,
482 m_sceneDepthCopyImage, VK_IMAGE_LAYOUT_TRANSFER_DST_OPTIMAL,
483 1, &region);
484
485 const VkImageMemoryBarrier after[] = {
486 MakeBarrier(m_hdrSceneImage, VK_IMAGE_ASPECT_COLOR_BIT, SINGLE_LAYER,
487 VK_IMAGE_LAYOUT_TRANSFER_SRC_OPTIMAL,
488 VK_IMAGE_LAYOUT_COLOR_ATTACHMENT_OPTIMAL, 0,
489 VK_ACCESS_COLOR_ATTACHMENT_READ_BIT |
490 VK_ACCESS_COLOR_ATTACHMENT_WRITE_BIT),
491 MakeBarrier(depthImage, depthAspects, SINGLE_LAYER,
492 VK_IMAGE_LAYOUT_TRANSFER_SRC_OPTIMAL,
493 VK_IMAGE_LAYOUT_DEPTH_STENCIL_ATTACHMENT_OPTIMAL, 0,
494 VK_ACCESS_DEPTH_STENCIL_ATTACHMENT_READ_BIT |
495 VK_ACCESS_DEPTH_STENCIL_ATTACHMENT_WRITE_BIT),
496 MakeBarrier(m_sceneColorCopyImage, VK_IMAGE_ASPECT_COLOR_BIT,
497 SINGLE_LAYER, VK_IMAGE_LAYOUT_TRANSFER_DST_OPTIMAL,
498 VK_IMAGE_LAYOUT_SHADER_READ_ONLY_OPTIMAL,
499 VK_ACCESS_TRANSFER_WRITE_BIT, VK_ACCESS_SHADER_READ_BIT),
500 MakeBarrier(m_sceneDepthCopyImage, depthAspects, SINGLE_LAYER,
501 VK_IMAGE_LAYOUT_TRANSFER_DST_OPTIMAL,
