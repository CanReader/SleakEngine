16/// Byte offsets of red, green, and blue inside one 4-byte texel; false when
17/// the format is not an 8-bit RGBA or BGRA layout.
18bool ChannelOffsets(VkFormat format, uint32_t& r, uint32_t& g, uint32_t& b) {
19 switch (format) {
20 case VK_FORMAT_B8G8R8A8_UNORM:
21 case VK_FORMAT_B8G8R8A8_SRGB:
22 r = 2;
23 g = 1;
24 b = 0;
25 return true;
26 case VK_FORMAT_R8G8B8A8_UNORM:
27 case VK_FORMAT_R8G8B8A8_SRGB:
28 case VK_FORMAT_A8B8G8R8_UNORM_PACK32:
29 case VK_FORMAT_A8B8G8R8_SRGB_PACK32:
30 r = 0;
31 g = 1;
32 b = 2;
33 return true;
34 default:
35 return false;
36 }
37}

39/// Picks a host-visible, host-coherent memory type, preferring cached ones.
40uint32_t FindReadbackMemoryType(VkPhysicalDevice gpu, uint32_t typeBits) {
41 VkPhysicalDeviceMemoryProperties props;
42 vkGetPhysicalDeviceMemoryProperties(gpu, &props);
43 const VkMemoryPropertyFlags coherent = VK_MEMORY_PROPERTY_HOST_VISIBLE_BIT |
44 VK_MEMORY_PROPERTY_HOST_COHERENT_BIT;
45 for (VkMemoryPropertyFlags wanted :
46 {coherent | VK_MEMORY_PROPERTY_HOST_CACHED_BIT, coherent}) {
47 for (uint32_t i = 0; i < props.memoryTypeCount; ++i) {
48 if ((typeBits & (1u << i)) &&
49 (props.memoryTypes[i].propertyFlags & wanted) == wanted)
50 return i;
51 }
52 }
53 return UINT32_MAX;
54}

58/// Arms a UI-free readback of the next presented frame into a binary PPM.
59bool VulkanRenderer::RequestFrameCapture(const std::string& path) {
60 if (path.empty()) return false;
61 if (!m_swapchainTransferSrc) {
62 SLEAK_ERROR(
63 "Frame capture: swapchain was created without TRANSFER_SRC usage");
64 return false;
65 }
66 uint32_t r, g, b;
67 if (!ChannelOffsets(scImageFormat, r, g, b)) {
68 SLEAK_ERROR("Frame capture: unsupported swapchain format {}",
69 static_cast<int>(scImageFormat));
70 return false;
71 }
72 m_capturePath = path;

157/// Waits for the captured frame, writes the PPM, and disarms the capture.
158void VulkanRenderer::FinishFrameCapture() {
159 m_captureRecorded = false;
160 std::string path;
161 path.swap(m_capturePath);
162
163 const VkResult wait = vkWaitForFences(
164 device, 1, &inFlightFences[currentFrame], VK_TRUE, 10'000'000'000ull);
165 void* mapped = nullptr;
166 if (wait != VK_SUCCESS ||
167 vkMapMemory(device, m_captureMemory, 0, VK_WHOLE_SIZE, 0, &mapped) !=
168 VK_SUCCESS) {
169 SLEAK_ERROR("Frame capture: readback of {} failed ({})", path,
170 static_cast<int>(wait));
171 CleanupFrameCapture();
172 return;
173 }
174
175 uint32_t r = 0, g = 1, b = 2;
176 ChannelOffsets(m_captureFormat, r, g, b);
177 const size_t pixels =
178 static_cast<size_t>(m_captureExtent.width) * m_captureExtent.height;
179 const auto* src = static_cast<const uint8_t*>(mapped);
180 std::vector<uint8_t> rgb(pixels * 3);
181 for (size_t i = 0; i < pixels; ++i) {
182 rgb[i * 3 + 0] = src[i * 4 + r];
183 rgb[i * 3 + 1] = src[i * 4 + g];
184 rgb[i * 3 + 2] = src[i * 4 + b];
185 }
186 vkUnmapMemory(device, m_captureMemory);
187 CleanupFrameCapture();
188
189 std::ofstream out(path, std::ios::binary);
190 out << "P6\n"
191 << m_captureExtent.width << " " << m_captureExtent.height << "\n255\n";
192 out.write(reinterpret_cast<const char*>(rgb.data()),
193 static_cast<std::streamsize>(rgb.size()));
194 if (!out) {
195 SLEAK_ERROR("Frame capture: could not write {}", path);
196 return;
197 }
198 SLEAK_INFO("Frame capture: wrote {}x{} to {}", m_captureExtent.width,
199 m_captureExtent.height, path);
200}
